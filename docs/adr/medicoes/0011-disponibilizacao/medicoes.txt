SELECT
  COALESCE(subq_6.metric_time__month, subq_12.metric_time__month) AS metric_time__month
  , MAX(subq_6.atos_publicados) AS atos_publicados
  , MAX(subq_12.mencoes_recebidas) AS mencoes_recebidas
FROM (
  SELECT
    metric_time__month
    , SUM(atos_publicados) AS atos_publicados
  FROM (
    SELECT
      DATE_TRUNC('month', data_publicacao) AS metric_time__month
      , origem AS ato__origem
      , 1 AS atos_publicados
    FROM "conselhos"."main"."fct_ato" atos_src_10000
  ) subq_2
  WHERE ato__origem = 'DOU'
  GROUP BY
    metric_time__month
) subq_6
FULL OUTER JOIN (
  SELECT
    metric_time__month
    , SUM(__mencoes_recebidas) AS mencoes_recebidas
  FROM (
    SELECT
      DATE_TRUNC('month', data_publicacao) AS metric_time__month
      , 1 AS __mencoes_recebidas
    FROM "conselhos"."main"."fct_mencao" mencoes_src_10000
  ) subq_10
  GROUP BY
    metric_time__month
) subq_12
ON
  subq_6.metric_time__month = subq_12.metric_time__month
GROUP BY
  COALESCE(subq_6.metric_time__month, subq_12.metric_time__month)
